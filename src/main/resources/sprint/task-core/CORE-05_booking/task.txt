## MỤC TIÊU
## ═══════════════════════════════════════════
Sinh viên đặt lịch gặp Giảng viên theo flow:
  Chọn Khoa → Chọn Giảng viên → Chọn ngày/giờ → Xác nhận

Hệ thống PHẢI chặn:
  (1) Đặt lịch vào khung giờ đã có người đặt (conflict)
  (2) Đặt lịch ngày/giờ trong quá khứ

## ═══════════════════════════════════════════
## DATABASE
## ═══════════════════════════════════════════

CREATE TABLE mentoring_sessions (
    id           BIGINT AUTO_INCREMENT PRIMARY KEY,
    student_id   BIGINT NOT NULL,
    lecturer_id  BIGINT NOT NULL,
    session_date DATE NOT NULL,
    start_time   TIME NOT NULL,
    end_time     TIME NOT NULL,
    status       ENUM('PENDING','CONFIRMED','COMPLETED','CANCELLED') NOT NULL DEFAULT 'PENDING',
    note         TEXT,
    created_at   DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (student_id) REFERENCES users(id),
    FOREIGN KEY (lecturer_id) REFERENCES users(id),
    -- Index để tối ưu conflict check:
    INDEX idx_lecturer_date (lecturer_id, session_date)
);

## BOOKING FLOW — 3 BƯỚC
## ═══════════════════════════════════════════

BƯỚC 1: GET /student/booking/step1
  - Lấy danh sách departments → model
  - Template: dropdown chọn Khoa

BƯỚC 2: GET /student/booking/step2?departmentId=X
  - Lấy danh sách lecturers theo department_id
  - Template: danh sách giảng viên (card hoặc table) để chọn

BƯỚC 3: GET /student/booking/step3?lecturerId=Y
  - Template: form chọn ngày, chọn slot giờ
  - Slot giờ gợi ý: 08:00–09:00, 09:00–10:00, ... 16:00–17:00
  - Khi chọn ngày → AJAX call GET /student/booking/available-slots?lecturerId=Y&date=Z
    → trả về JSON danh sách slot CHƯA bị đặt

BƯỚC 4: POST /student/booking/confirm
  - Nhận BookingRequestDto
  - Service validate + save → redirect /student/booking/success

## LOGIC CONFLICT CHECK (QUAN TRỌNG NHẤT)
## ═══════════════════════════════════════════

// Trong MentoringSessionRepository:
@Query("""
    SELECT COUNT(ms) > 0 FROM MentoringSession ms
    WHERE ms.lecturer.id = :lecturerId
      AND ms.sessionDate = :date
      AND ms.startTime < :endTime
      AND ms.endTime > :startTime
      AND ms.status NOT IN (com.platform.entity.SessionStatus.CANCELLED)
""")
boolean existsConflict(@Param("lecturerId") Long lecturerId,
                        @Param("date") LocalDate date,
                        @Param("startTime") LocalTime startTime,
                        @Param("endTime") LocalTime endTime);

// Trong BookingService.createBooking():
// Bước 1: Kiểm tra quá khứ
LocalDateTime bookingDateTime = LocalDateTime.of(dto.getSessionDate(), dto.getStartTime());
if (bookingDateTime.isBefore(LocalDateTime.now())) {
    throw new BusinessException("Không thể đặt lịch trong quá khứ");
}
// Bước 2: Kiểm tra conflict
if (sessionRepo.existsConflict(dto.getLecturerId(), dto.getSessionDate(),
                                dto.getStartTime(), dto.getEndTime())) {
    throw new SlotConflictException("Khung giờ này đã có lịch hẹn khác");
}
// Bước 3: Save nếu OK
MentoringSession session = new MentoringSession();
// ... set fields ...
session.setStatus(SessionStatus.PENDING);
return sessionRepo.save(session);
