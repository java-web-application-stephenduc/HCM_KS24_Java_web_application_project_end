## MỤC TIÊU
## ═══════════════════════════════════════════
Sinh viên xem lại lịch sử toàn bộ các lần tư vấn.
Mỗi record phải hiển thị "Hồ sơ học tập" đầy đủ:
  - Tên giảng viên + Khoa
  - Ngày/giờ buổi tư vấn
  - Điểm đánh giá + Nhận xét (nếu đã hoàn thành)
  - Danh sách thiết bị đã mượn + số lượng (nếu có)

Đây là yêu cầu JOIN nhiều bảng — KHÔNG được loop trong Java để lấy data.

## ═══════════════════════════════════════════
## QUERY CẦN VIẾT
## ═══════════════════════════════════════════

-- Full JOIN để lấy hồ sơ một lần:
SELECT
    ms.id            AS session_id,
    ms.session_date,
    ms.start_time,
    ms.end_time,
    ms.status,
    up.full_name     AS lecturer_name,
    d.name           AS department_name,
    ae.score,
    ae.feedback,
    br.id            AS record_id,
    br.status        AS borrow_status,
    e.name           AS equipment_name,
    bd.quantity
FROM mentoring_sessions ms
    JOIN users u            ON ms.lecturer_id = u.id
    JOIN user_profiles up   ON u.id = up.user_id
    JOIN lecturers l        ON u.id = l.user_id
    JOIN departments d      ON l.department_id = d.id
    LEFT JOIN academic_evaluations ae ON ms.id = ae.session_id
    LEFT JOIN borrowing_records br    ON ms.id = br.session_id
    LEFT JOIN borrowing_details bd    ON br.id = bd.record_id
    LEFT JOIN equipments e            ON bd.equipment_id = e.id
WHERE ms.student_id = :studentId
ORDER BY ms.session_date DESC, ms.start_time DESC;

## DATA GROUPING LOGIC (Java Service)
## ═══════════════════════════════════════════

// Flat rows từ DB → Map<Long, AcademicRecordDto>
// Vì LEFT JOIN borrowing_details có thể sinh nhiều row cho 1 session

public List<AcademicRecordDto> getAcademicHistory(Long studentId) {
    List<AcademicHistoryProjection> rows = sessionRepo.findAcademicHistory(studentId);

    // Group rows by sessionId
    Map<Long, AcademicRecordDto> grouped = new LinkedHashMap<>();
    for (AcademicHistoryProjection row : rows) {
        AcademicRecordDto dto = grouped.computeIfAbsent(row.getSessionId(), id ->
            AcademicRecordDto.builder()
                .sessionId(row.getSessionId())
                .sessionDate(row.getSessionDate())
                .lecturerName(row.getLecturerName())
                .score(row.getScore())
                .feedback(row.getFeedback())
                .equipments(new ArrayList<>())
                .build()
        );
        // Thêm equipment nếu có (LEFT JOIN có thể null)
        if (row.getEquipmentName() != null) {
            dto.getEquipments().add(new BorrowedEquipmentDto(
                row.getEquipmentName(), row.getQuantity()
            ));
        }
    }
    return new ArrayList<>(grouped.values());
}
