## 4. ENTITY PATTERN (Hibernate JPA)
## ═══════════════════════════════════════════════

// Mọi Entity đều kế thừa pattern này:
@Entity
@Table(name = "table_name")
@Getter @Setter @NoArgsConstructor  // Lombok
public class EntityName {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    // FK Many-to-One:
    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "fk_column_id", nullable = false)
    private OtherEntity otherEntity;

    // Timestamp tự động:
    @Column(name = "created_at", updatable = false)
    @CreationTimestamp
    private LocalDateTime createdAt;

    // Enum mapping:
    @Enumerated(EnumType.STRING)
    @Column(nullable = false)
    private StatusEnum status;
}

QUAN TRỌNG:
  - KHÔNG dùng FetchType.EAGER cho collection (gây N+1 query)
  - KHÔNG expose Entity ra ngoài Controller → luôn dùng DTO
  - Equals/hashCode chỉ dựa vào id (tránh lazy load trong collection)

## 6. CONTROLLER PATTERN
## ═══════════════════════════════════════════════

@Controller
@RequestMapping("/student")
@RequiredArgsConstructor
public class StudentController {

    private final BookingService bookingService;

    // GET → render view
    @GetMapping("/booking")
    public String showBookingForm(Model model, @AuthenticationPrincipal UserDetails user) {
        model.addAttribute("bookingForm", new BookingFormDto());
        model.addAttribute("lecturers", bookingService.getAvailableLecturers());
        return "student/booking-form";   // → templates/student/booking-form.html
    }

    // POST → redirect after success (PRG pattern)
    @PostMapping("/booking")
    public String submitBooking(@Valid @ModelAttribute BookingFormDto form,
                                 BindingResult errors,
                                 RedirectAttributes flash,
                                 @AuthenticationPrincipal UserDetails user) {
        if (errors.hasErrors()) return "student/booking-form";
        bookingService.createBooking(form, user.getUsername());
        flash.addFlashAttribute("success", "Đặt lịch thành công!");
        return "redirect:/student/dashboard";
    }
}

## 7. SECURITY CONFIGURATION
## ═══════════════════════════════════════════════

@Configuration
@EnableWebSecurity
public class SecurityConfig {

    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        http
            .authorizeHttpRequests(auth -> auth
                .requestMatchers("/auth/**", "/css/**", "/js/**").permitAll()
                .requestMatchers("/admin/**").hasRole("ADMIN")
                .requestMatchers("/lecturer/**").hasRole("LECTURER")
                .requestMatchers("/student/**").hasRole("STUDENT")
                .anyRequest().authenticated()
            )
            .formLogin(form -> form
                .loginPage("/auth/login")
                .defaultSuccessUrl("/dashboard", true)
                .failureUrl("/auth/login?error")
            )
            .logout(logout -> logout.logoutUrl("/auth/logout").logoutSuccessUrl("/auth/login"));
        return http.build();
    }

    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();
    }
}

// Pattern chuẩn cho CORE-06 (lưu nhiều bảng cùng lúc):
@Transactional
public void completeCounselingSession(Long sessionId, EvaluationDto evalDto) {
    // Step 1: Load session, validate status
    MentoringSession session = sessionRepo.findById(sessionId)
        .orElseThrow(() -> new ResourceNotFoundException("Session not found"));
    if (session.getStatus() != SessionStatus.CONFIRMED)
        throw new BusinessException("Session is not in CONFIRMED state");

    // Step 2: Update session status
    session.setStatus(SessionStatus.COMPLETED);
    sessionRepo.save(session);

    // Step 3: Save evaluation
    AcademicEvaluation eval = mapper.toEntity(evalDto, session);
    evaluationRepo.save(eval);

    // Step 4: Create borrowing record + details
    BorrowingRecord record = new BorrowingRecord(session, BorrowingStatus.PENDING_DISPATCH);
    borrowingRepo.save(record);

    List<BorrowingDetail> details = evalDto.getEquipmentItems().stream()
        .map(item -> new BorrowingDetail(record, item.getEquipmentId(), item.getQuantity()))
        .toList();
    borrowingDetailRepo.saveAll(details);

    // Nếu bất kỳ bước nào throw RuntimeException → Spring tự Rollback toàn bộ
}

@Transactional
public void dispatchEquipment(Long recordId) {
    BorrowingRecord record = borrowingRepo.findById(recordId).orElseThrow(...);
    List<BorrowingDetail> details = borrowingDetailRepo.findByRecordId(recordId);

    // Kiểm tra tất cả trước khi trừ (all-or-nothing)
    for (BorrowingDetail detail : details) {
        Equipment eq = equipmentRepo.findByIdWithLock(detail.getEquipment().getId()); // SELECT FOR UPDATE
        if (eq.getQuantityAvailable() < detail.getQuantity()) {
            throw new InsufficientStockException("Thiết bị [" + eq.getName() + "] không đủ tồn kho");
        }
    }
    // Trừ kho chỉ khi tất cả OK
    for (BorrowingDetail detail : details) {
        Equipment eq = equipmentRepo.findById(detail.getEquipment().getId()).orElseThrow();
        eq.setQuantityAvailable(eq.getQuantityAvailable() - detail.getQuantity());
        equipmentRepo.save(eq);
    }
    record.setStatus(BorrowingStatus.DISPATCHED);
    borrowingRepo.save(record);
}

// Repository — lock row để tránh race condition:
@Lock(LockModeType.PESSIMISTIC_WRITE)
@Query("SELECT e FROM Equipment e WHERE e.id = :id")
Optional<Equipment> findByIdWithLock(@Param("id") Long id);
